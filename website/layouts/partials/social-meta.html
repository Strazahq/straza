{{- /* Open Graph and Twitter card tags. They are written here rather than taken from Hugo's embedded templates, which use the plain title, so three pages would share "Policies" when a link to them is shared. The title is the one carries, without the site suffix. There is no og:image, because static/ holds no image made for sharing. */ -}} {{- $title := cond .IsHome site.Title (.Params.seo_title | default .Title) -}} <meta property="og:site_name" content="{{ site.Title }}"> <meta property="og:type" content="{{ if .IsPage }}article{{ else }}website{{ end }}"> <meta property="og:title" content="{{ $title }}"> <meta property="og:url" content="{{ .Permalink }}"> {{- with (.Params.seo_description | default .Description) }} <meta property="og:description" content="{{ . }}"> {{- end }} {{- if and .IsPage (not .Lastmod.IsZero) }} <meta property="article:modified_time" content="{{ .Lastmod.Format "2006-01-02T15:04:05Z07:00" }}"> {{- end }} <meta name="twitter:card" content="summary"> <meta name="twitter:title" content="{{ $title }}"> {{- with (.Params.seo_description | default .Description) }} <meta name="twitter:description" content="{{ . }}"> {{- end -}}
