{{- /* A picture the console screenshot pipeline does not make, such as a screen of another product: {{< picture src="pictures/entra-role-groups/groups-list.png" alt="..." caption="..." >}} A console screen in both themes names dark and light in place of src, and site.css shows the one that matches the theme. A file that is missing, or a picture without alt, fails the build. */ -}} {{- $dark := .Get "dark" | default (.Get "src") -}} {{- $light := .Get "light" -}} {{- $alt := .Get "alt" -}} {{- if not $dark }}{{ errorf "%s: picture has neither src nor dark. Name the file under website/static." .Position }}{{ end -}} {{- if not $alt }}{{ errorf "%s: picture %q has no alt. Say what the picture shows." .Position $dark }}{{ end -}} {{- range slice $dark $light -}} {{- if and . (not (fileExists (printf "static/%s" .))) }}{{ errorf "%s: picture names website/static/%s, which does not exist." $.Position . }}{{ end -}} {{- end -}} {{- $size := images.Config (printf "static/%s" $dark) -}}
